ALIVE 빈자리,
열리면 소리와 윈도우 알림으로.

설치 없이 크롬 북마크로 시작하는 PC용 알림 도구입니다. 알림 기능 수정판: 기존 북마크를 지우고 아래 버튼으로 다시 등록하세요.

2026년 10월 24일 · 31일모든 시간대기본 2분 · 1/2/3/5/10/20/30/40분 선택

① 북마크를 등록하세요

  1. 크롬에서 이 파일을 열고 Ctrl + Shift + B를 눌러 북마크바를 표시합니다.
  2. 아래 초록 버튼을 크롬 상단의 북마크바로 끌어다 놓으세요.
    ALIVE 빈자리 알림등록된 북마크 이름은 그대로 두면 됩니다.

② 예약 페이지에서 실행하세요

  1. 제로월드 홍대점 예약 페이지 열기 ↗
  2. 예약 페이지가 열린 상태에서, 방금 등록한 「ALIVE 빈자리 알림」 북마크를 누릅니다.
  3. 오른쪽 위에 뜨는 패널에서 「윈도우 알림 켜기」를 누르고 크롬의 알림 요청에서 허용을 선택합니다. 예약 사이트에 대한 알림 권한입니다.
  4. 「소리+알림 시험」으로 알림음과 윈도우 알림을 확인한 뒤 「감시 시작」을 누릅니다.
  5. 첫 확인이 끝나면 두 날짜의 상태와 마지막 확인 시각이 표시됩니다. 빈자리를 찾으면 알림음·윈도우 알림·화면 표시·탭 제목으로 알려줍니다.

사용하는 동안

• 윈도우 알림에는 발견한 날짜와 시간이 표시됩니다. 알림을 누르면 감시 페이지로 돌아갑니다. 알림이 허용되어 있어도 윈도우의 방해 금지 모드나 크롬 알림 설정에 따라 배너가 보이지 않을 수 있으므로 시험 알림으로 확인하세요.

• 예약 페이지를 별도 크롬 창에 보이게 두고, PC와 소리를 켜두세요. 절전·탭 비활성화·메모리 절약·네트워크 끊김은 확인을 지연시키거나 중단할 수 있습니다. 설정 간격은 실시간 보장 간격이 아닙니다.

• 페이지를 새로고침하거나 다른 페이지로 이동하면 감시가 종료됩니다. 다시 북마크를 눌러 시작하세요.

• 빈자리는 발견 직후에도 다시 마감될 수 있습니다. 「예약 페이지 열기」에서 해당 날짜 → ALIVE를 선택해 직접 예약하세요.

• 「알림음 끄기」는 소리만 멈춥니다. 「중지」는 감시를 멈추고, ×는 패널을 닫습니다.

• 확인 실패는 매진으로 표시하지 않습니다. 연속 실패 시 간격을 늘리고, 3회 연속 실패하면 중지합니다. 10월 31일이 지나면 자동 종료합니다.

• 개인정보 입력과 예약 신청은 수행하지 않습니다. 이 도구가 별도 서버로 보내는 데이터는 없으며, 공식 예약 화면을 불러와 확인합니다.

끌어놓기가 안 될 때: 북마크 주소 복사

아래 「주소 복사」를 누른 뒤, Ctrl + Shift + O로 북마크 관리자를 여세요. 빈 공간을 우클릭해 새 북마크 추가를 선택하고, 이름은 「ALIVE 빈자리 알림」, URL에는 복사한 주소를 붙여넣으세요.

    if(keepAwake) {keepAwake.release().catch(()=>{}); keepAwake=null;}
  }
  function schedule() {
    if(!running) return;
    const seconds=retrySeconds(Number(el('period').value),failures);
    nextAt=Date.now()+seconds*1000;
    clearTimeout(timer); timer=setTimeout(cycle,seconds*1000);
  }
  async function cycle() {
    if(!running || busy) return;
    busy=true; const g=generation; let errors=0; const found=[];
    clearTimeout(timer);
    try {
      const dates=DATES.filter(d=>future(d));
      for(const date of DATES.filter(d=>!future(d))) show(date,'대상 날짜 종료');
      if(!dates.length) {stop('두 날짜가 모두 지나 감시를 종료했습니다.'); return;}
      for(const date of dates) {
        valid(g); setStatus(date.slice(5).replace('-','/')+' ALIVE 확인 중…');
        try {
          const available=await inspect(date,g); valid(g);
          const fresh=newSlots(seen.get(date)||[],available); seen.set(date,available);
          show(date,available.length?available.join(' · '):'현재 예약 가능 시간 없음',available.length?'available':'');
          log(date.slice(5)+' '+(available.length?available.join(', '):'빈자리 없음'));
          if(fresh.length) {const msg=date.slice(5).replace('-','/')+' '+fresh.join(' · '); found.push(msg); notify(found);}
        } catch(e) {
          valid(g);
          if(e.blocked) {stop(e.message); beep(); return;}
          errors++; show(date,'확인 실패: '+e.message,'error'); log(date.slice(5)+' 확인 실패');
        }
        await sleep(900);
      }
      valid(g); failures=errors?failures+1:0; lastFinished=Date.now();
      if(failures>=3) {stop('3회 연속 확인에 실패해 중지했습니다. 연결 상태 확인 후 다시 시작하세요.'); beep(); return;}
      schedule();
    } catch(e) { if(running && generation===g) {stop('확인을 중지했습니다: '+e.message); beep();} }
    finally {busy=false;}
  }
  el('start').onclick=()=>{
    if(running || busy) return;
    ensureAudio(); running=true; generation++; failures=0; lastWake=Date.now(); el('start').disabled=true; el('stop').disabled=false;
    // Screen wake lock is best effort. It does not change any OS power settings.
    if(navigator.wakeLock && document.visibilityState==='visible') navigator.wakeLock.request('screen').then(w=>{if(running) keepAwake=w; else w.release();}).catch(()=>{});
    log('감시 시작');
    countdown=setInterval(()=>{
      if(!running) return;
      const now=Date.now();
      if(now-lastWake>90000) {log('절전 또는 브라우저 지연 감지'); if(!busy && now>=nextAt) cycle();}
      lastWake=now;
      if(!busy) {
        const left=Math.max(0,Math.ceil((nextAt-now)/1000));
        setStatus('마지막 확인 '+(lastFinished?new Intl.DateTimeFormat('ko-KR',{timeZone:'Asia/Seoul',hour:'2-digit',minute:'2-digit',second:'2-digit'}).format(lastFinished):'—')+'\n다음 확인 약 '+(left>=60?Math.floor(left/60)+'분 ':'')+left%60+'초 후'+(failures?' · 확인 실패 재시도':''));
      }
    },1000);
    cycle();
  };
  el('stop').onclick=()=>stop();
  el('close').onclick=()=>{stop(); host.remove(); document.removeEventListener('visibilitychange',visible); if(audio) audio.close().catch(()=>{});};

    } catch(_) {el('notification-status').textContent='알림 권한을 설정하지 못했습니다. 사이트 설정에서 확인하세요.';}
  };
  el('test').onclick=()=>{
    ensureAudio(); setTimeout(beep,180);
    sendDesktop('ALIVE 시험 알림 — 빈자리 발견 아님','소리와 알림 표시 확인용입니다. 실제 빈자리 알림에는 날짜와 시간이 표시됩니다.');
    log('소리+알림 시험');
  };
  el('ack').onclick=acknowledge;
  el('book').onclick=()=>window.open(URL,'_blank','noopener,noreferrer');
  function visible() {if(running&&document.visibilityState==='visible'&&!busy&&Date.now()>=nextAt) cycle();}
  document.addEventListener('visibilitychange',visible);
